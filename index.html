async function uploadVideo(event) { event.preventDefault(); const spinner = document.getElementById('loading-spinner'); // Aapke spinner ki ID spinner.style.display = 'block'; // Gol daira shuru const formData = new FormData(document.getElementById('upload-form')); try { let response = await fetch('/upload', { method: 'POST', body: formData }); let result = await response.json(); if (result.success) { // Feed par redirect ya naya video feed mein add karein location.reload(); } else { alert('Upload fail ho gaya: ' + result.error); } } catch (err) { console.error('Error:', err); alert('Kuch masla ho gaya, dobara koshish karein.'); } finally { // Yeh block chalega hi chalega, chahe success ho ya error, gol daira gayab! spinner.style.display = 'none'; } }
